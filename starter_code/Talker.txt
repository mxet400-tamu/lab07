#!/usr/bin/env python3
# This line tells Linux this is a Python3 script

# ===== IMPORT SECTION =====
import rclpy  # Import the ROS2 Python library
from rclpy.node import Node  # Import the Node class - the base for all ROS2 nodes
from std_msgs.msg import String  # Import the String message type


# ===== CLASS DEFINITION =====
class MinimalPublisher(Node):
    """
    A simple publisher node that sends "Hello World" messages.
    """
    
    def __init__(self):
        """
        Constructor: This runs when you create a MinimalPublisher object.
        It sets up the publisher and timer.
        """
        # Call the parent class (Node) constructor and name this node
        super().__init__('minimal_publisher')
        
        # Create a publisher object:
        # - It will publish String messages
        # - On the topic named 'topic'
        # - With a queue size of 10 (holds up to 10 messages if sending faster than network can handle)
        self.publisher_ = self.create_publisher(String, 'topic', 10)
        
        # Create a timer that calls timer_callback every 0.5 seconds
        timer_period = 0.5  # seconds
        self.timer = self.create_timer(timer_period, self.timer_callback)
        
        # Initialize a counter variable
        self.i = 0
        
        # Print a message to confirm the node started
        self.get_logger().info('Talker node has been started!')
    
    def timer_callback(self):
        """
        This function runs every 0.5 seconds (because of the timer).
        It creates a message, publishes it, and logs it to the console.
        """
        # Create a new String message object
        msg = String()
        
        # Set the message data (the actual content)
        msg.data = 'Hello World: %d' % self.i
        
        # Publish the message on the topic
        self.publisher_.publish(msg)
        
        # Log the message to the terminal so you can see it
        self.get_logger().info('Publishing: "%s"' % msg.data)
        
        # Increment the counter for the next message
        self.i += 1


# ===== MAIN FUNCTION =====
def main(args=None):
    """
    The main function that runs when you execute this script.
    """
    # Initialize the ROS2 Python library
    rclpy.init(args=args)
    
    # Create an instance of our MinimalPublisher class
    minimal_publisher = MinimalPublisher()
    
    # Keep the node running and processing callbacks (like timer_callback)
    # This is a blocking call - the program stays here until you press Ctrl+C
    rclpy.spin(minimal_publisher)
    
    # Clean up: Destroy the node explicitly (optional, but good practice)
    minimal_publisher.destroy_node()
    
    # Shutdown the ROS2 Python library
    rclpy.shutdown()


# ===== SCRIPT ENTRY POINT =====
if __name__ == '__main__':
    # If this script is run directly (not imported), execute main()
    main()
